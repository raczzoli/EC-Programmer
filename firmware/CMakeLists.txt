cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH to the pico-sdk directory")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_ec_probe C CXX ASM)
pico_sdk_init()
add_executable(pico_ec_probe main.c)
target_link_libraries(pico_ec_probe pico_stdlib hardware_gpio hardware_pio hardware_dma hardware_clocks)
pico_enable_stdio_usb(pico_ec_probe 1)
pico_enable_stdio_uart(pico_ec_probe 0)
pico_add_extra_outputs(pico_ec_probe)
