cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH to the pico-sdk directory")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_ec_probe C CXX ASM)
pico_sdk_init()
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/../VERSION" EC_VERSION LIMIT_COUNT 1)
if(NOT DEFINED EC_BUILD_ID)
  set(EC_BUILD_ID "local")
endif()
configure_file(version.h.in generated/version.h @ONLY)
add_executable(pico_ec_probe main.c ite_bus.c ite_dbgr.c ite_session.c ite_flash.c usb_commands.c usb_output.c ene_edi.c)
target_include_directories(pico_ec_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(pico_ec_probe PRIVATE -Wall -Wextra)
pico_set_program_name(pico_ec_probe "EC Programmer")
pico_set_program_version(pico_ec_probe "${EC_VERSION}-${EC_BUILD_ID}")
target_link_libraries(pico_ec_probe pico_stdlib hardware_gpio hardware_pio hardware_dma hardware_clocks hardware_spi)
pico_enable_stdio_usb(pico_ec_probe 1)
pico_enable_stdio_uart(pico_ec_probe 0)
pico_add_extra_outputs(pico_ec_probe)
